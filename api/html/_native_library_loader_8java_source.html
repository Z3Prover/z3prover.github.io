125 private static String hash(InputStream input) throws IOException {
126 MessageDigest digest = digest();
127 byte[] buffer = new byte[65536];
128 int count;
129 while ((count = input.read(buffer)) != -1) {
130 digest.update(buffer, 0, count);
131 }
132 StringBuilder text = new StringBuilder(64);
133 for (byte value : digest.digest()) {
134 text.append(Character.forDigit((value & 255) >>> 4, 16));
135 text.append(Character.forDigit(value & 15, 16));
136 }
137 return text.toString();
138 }

157 private static Path cacheRoot() throws IOException {
158 String configured = System.getProperty("z3.nativeCacheDir");
159 String profile = System.getProperty("user.home");
160 if (profile == null || profile.isEmpty()) {
161 throw new IOException("No user.home for the native cache");
162 }
163 Path root = (configured == null
164 ? Paths.get(profile, ".z3", "native")
165 : Paths.get(configured)).toAbsolutePath().normalize();
166 checkPath(root);
167 if (root.getFileSystem().supportedFileAttributeViews().contains("posix")) {
168 Files.createDirectories(root, PosixFilePermissions.asFileAttribute(
169 PosixFilePermissions.fromString("rwx------")));
170 Set<PosixFilePermission> permissions = Files.getPosixFilePermissions(root);
171 if (permissions.contains(PosixFilePermission.GROUP_WRITE)
172 || permissions.contains(PosixFilePermission.OTHERS_WRITE)) {
173 throw new IOException("Native cache is writable by other users: " + root);
174 }
175 } else {
176 // On Windows inherit the user's profile ACL, including for .z3/native.
177 Files.createDirectories(root);
178 }
179 checkPath(root);
180 if (!Files.getOwner(root).equals(Files.getOwner(Paths.get(System.getProperty("user.home"))))) {
181 throw new IOException("Native cache is not owned by the profile owner: " + root);
182 }
183 return root;
184 }

194 private static void loadCachedLocked(String platform, String core, String bridge) throws IOException {
195 String coreHash;
196 String bridgeHash;
197 // Derive identity from the actual bundled bytes, not the version or a mutable manifest.
198 try (InputStream input = resource(platform, core)) {
199 coreHash = hash(input);
200 }
201 try (InputStream input = resource(platform, bridge)) {
202 bridgeHash = hash(input);
203 }
204 String identity;
205 try (InputStream input = new java.io.ByteArrayInputStream(
206 (coreHash + bridgeHash).getBytes(java.nio.charset.StandardCharsets.US_ASCII))) {
207 identity = hash(input);
208 }
209 Path root = cacheRoot();
210 // One process lock covers verification, publication, loading and eviction.
211 // File locks overlap across ClassLoaders in one JVM; that case uses private copies.
212 try (FileChannel channel = FileChannel.open(root.resolve(".lock"),
213 StandardOpenOption.CREATE, StandardOpenOption.WRITE, LinkOption.NOFOLLOW_LINKS);
214 FileLock lock = channel.lock()) {
215 Path directory = root.resolve(platform + "-" + identity);
216 checkPath(directory);
217 if (!matches(directory.resolve(core), coreHash) || !matches(directory.resolve(bridge), bridgeHash)) {
218 if (Files.exists(directory, LinkOption.NOFOLLOW_LINKS)) {
219 removeEntry(directory, core, bridge);
220 }
221 Path staging = Files.createTempDirectory(root, ".extract-");
222 try {
223 try (InputStream input = resource(platform, core)) {
224 Files.copy(input, staging.resolve(core));
225 }
226 try (InputStream input = resource(platform, bridge)) {
227 Files.copy(input, staging.resolve(bridge));
228 }
229 if (!matches(staging.resolve(core), coreHash) || !matches(staging.resolve(bridge), bridgeHash)) {
230 throw new IOException("Bundled native resources changed during extraction");
231 }
232 // Publish only complete, verified pairs. If atomic moves are unavailable,
233 // preserve the old private-extraction behavior instead of a partial cache.
234 Files.move(staging, directory, StandardCopyOption.ATOMIC_MOVE);
235 } finally {
236 if (Files.exists(staging, LinkOption.NOFOLLOW_LINKS)) {
237 removeEntry(staging, core, bridge);
238 }
239 }
240 }
241 Files.setLastModifiedTime(directory, java.nio.file.attribute.FileTime.fromMillis(System.currentTimeMillis()));
242 System.load(directory.resolve(core).toString());
243 System.load(directory.resolve(bridge).toString());
244 prune(root, directory, platform, core, bridge);
245 }

258 throw new IOException("Unexpected native cache entry: " + file);

271 private static void prune(Path root, Path current, String platform, String core, String bridge) {
272 try {
273 TreeMap<String, Path> previous = new TreeMap<String, Path>();
274 long stale = System.currentTimeMillis() - 30L * 24 * 60 * 60 * 1000;
275 try (DirectoryStream<Path> entries = Files.newDirectoryStream(root)) {
276 for (Path entry : entries) {
277 String name = entry.getFileName().toString();
278 if (entry.equals(current) || !Files.isDirectory(entry, LinkOption.NOFOLLOW_LINKS)) {
279 continue;
280 }
281 long modified = Files.getLastModifiedTime(entry, LinkOption.NOFOLLOW_LINKS).toMillis();
282 if (name.matches(platform + "-[0-9a-f]{64}")) {
283 try { checkEntry(entry, core, bridge); } catch (IOException ignored) { continue; }
284 previous.put(String.format(Locale.ROOT, "%020d-%s", modified, name), entry);
285 } else if (name.startsWith(".extract-") && modified < stale) {
286 try { removeEntry(entry, core, bridge); } catch (IOException ignored) { }
287 }
288 }
289 }
290 // Keep this pair and the two most recently used pairs for this platform.
291 while (previous.size() > 2) {
292 Path entry = previous.pollFirstEntry().getValue();
293 try { removeEntry(entry, core, bridge); } catch (IOException ignored) { }
294 }
295 } catch (IOException | SecurityException ignored) {
296 // A loaded DLL may remain locked on Windows. Cleanup cannot break a working load.
297 }
298 }

307 static String platform() {
308 String os = System.getProperty("os.name", "").toLowerCase(Locale.ROOT);
309 String arch = System.getProperty("os.arch", "").toLowerCase(Locale.ROOT);
310 String osPart;
311 if (os.startsWith("windows")) {
312 osPart = "win";
313 } else if (os.startsWith("mac") || os.equals("darwin")) {
314 osPart = "osx";
315 } else if (os.equals("linux")) {
316 osPart = "linux";
317 } else {
318 osPart = "unsupported-os";
319 }
320 String archPart;
321 if (arch.equals("amd64") || arch.equals("x86_64")) {
322 archPart = "x64";
323 } else if (arch.equals("aarch64") || arch.equals("arm64")) {
324 archPart = "arm64";
325 } else if (arch.equals("x86") || arch.matches("i[3-6]86")) {
326 archPart = "x86";
327 } else {
328 archPart = "unsupported-arch";
329 }
330 return osPart + "-" + archPart;
