529 public DatatypeSort[] MkDatatypeSorts(Symbol[] names, Constructor[][] c)
530 {
531 Debug.Assert(names != null);
532 Debug.Assert(c != null);
533 Debug.Assert(names.Length == c.Length);
534 //Debug.Assert(Contract.ForAll(0, c.Length, j => c[j] != null));
535 Debug.Assert(names.All(name => name != null));
536
537 CheckContextMatch<Symbol>(names);
538 uint n = (uint)names.Length;
539 ConstructorList[] cla = new ConstructorList[n];
540 IntPtr[] n_constr = new IntPtr[n];
541 for (uint i = 0; i < n; i++)
542 {
543 Constructor[] constructor = c[i];
544 CheckContextMatch<Constructor>(constructor);
545 cla[i] = new ConstructorList(this, constructor);
546 n_constr[i] = cla[i].NativeObject;
547 }
548 IntPtr[] n_res = new IntPtr[n];
549 Native.Z3_mk_datatypes(nCtx, n, Symbol.ArrayToNative(names), n_res, n_constr);
550 DatatypeSort[] res = new DatatypeSort[n];
551 for (uint i = 0; i < n; i++)
552 res[i] = new DatatypeSort(this, n_res[i]);
553 return res;
554 }
